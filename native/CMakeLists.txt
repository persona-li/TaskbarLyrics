cmake_minimum_required(VERSION 3.24)
project(TaskbarLyricsNative VERSION 1.0.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use MSVC on Windows with the Windows SDK and C++/WinRT")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../artifacts/native-deps" CACHE PATH "Restored native dependencies")
add_library(native_core INTERFACE)
target_include_directories(native_core INTERFACE src "${DEPS}/json")
target_compile_definitions(native_core INTERFACE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(native_core INTERFACE /utf-8 /W4 /permissive- /EHsc)
add_executable(TaskbarLyrics.Native WIN32 src/main.cpp src/media.cpp src/overlay.cpp src/library.cpp src/resources.rc)
target_link_libraries(TaskbarLyrics.Native PRIVATE native_core windowsapp d2d1 dwrite uiautomationcore dwmapi shell32 shlwapi comdlg32 winhttp crypt32 ole32 advapi32 gdi32 gdiplus "${DEPS}/webview/build/native/x64/WebView2LoaderStatic.lib")
target_include_directories(TaskbarLyrics.Native PRIVATE "${DEPS}/webview/build/native/include")
target_link_options(TaskbarLyrics.Native PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest)
enable_testing()
add_executable(native_tests tests/core_tests.cpp)
target_link_libraries(native_tests PRIVATE native_core)
add_test(NAME native_invariants COMMAND native_tests)
add_executable(native_storage_tests tests/storage_tests.cpp src/library.cpp)
target_link_libraries(native_storage_tests PRIVATE native_core winhttp)
add_test(NAME native_storage_isolation COMMAND native_storage_tests)
add_executable(native_platform_tests tests/platform_policy_tests.cpp)
target_link_libraries(native_platform_tests PRIVATE native_core)
add_test(NAME native_platform_policy COMMAND native_platform_tests)
add_executable(native_settings_tests tests/settings_tests.cpp)
target_link_libraries(native_settings_tests PRIVATE native_core advapi32)
add_test(NAME native_settings_migration COMMAND native_settings_tests)
add_executable(native_lyrics_tests tests/lyrics_tests.cpp src/library.cpp)
target_link_libraries(native_lyrics_tests PRIVATE native_core winhttp)
add_test(NAME native_lyrics_parity COMMAND native_lyrics_tests)
add_executable(native_tray_tests tests/tray_menu_tests.cpp)
target_link_libraries(native_tray_tests PRIVATE native_core gdiplus shell32 advapi32 gdi32)
add_test(NAME native_tray_dismissal COMMAND native_tray_tests)
